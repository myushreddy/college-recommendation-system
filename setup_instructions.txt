    state, 
    tier, 
    nirf_rank, 
    overall_score

LIMIT 10;

--- Query 3: Check facilities ---
SELECT 
    c.college_name,
    f.has_hostel,
    f.has_library,
    f.has_gym
FROM colleges c
INNER JOIN college_facilities f ON c.college_id = f.college_id
LIMIT 10;

--- Query 4: CS courses with fees ---
SELECT 
    c.college_name,
    co.course_name,
    co.fee_per_year,
    co.course_category
FROM courses co
INNER JOIN colleges c ON co.college_id = c.college_id
WHERE co.is_cs_related = TRUE
ORDER BY co.fee_per_year ASC NULLS LAST
LIMIT 10;
